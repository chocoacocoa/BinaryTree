using System;

public class BSnode
{
    public int data;
    public BSnode? left;
    public BSnode? right;

    public BSnode(int data)
    {
        this.data = data;
        left = null;
        right = null;
    }
}

class Program
{
    public static BSnode? root;

    public static BSnode? insertRecursion(BSnode? current, int data)
    {
        if(current == null)
        {
            return new BSnode(data);
        }

        if(data < current.data)
        {
            current.left = insertRecursion(current.left, data);
        }
        else if (data > current.data)
        {
            current.right = insertRecursion(current.right, data);
        }

        return current;
    }

    public static bool searchRecursion(BSnode? current, int target)
    {
        if(current == null)
        {
            return false;
        }

        if (current.data == target)
        {
            return true;
        }

        if (target < current.data)
        {
            return searchRecursion(current.left, target);
        }
        else
        {
            return searchRecursion(current.right, target);
        }
    }

    public static void printInOrder(BSnode? node)
    {
        if(node != null)
        {
            printInOrder(node.left);
            Console.Write(node.data + " ");
            printInOrder(node.right);
        }
    }

    static void Main(string[] args)
    {
        Console.WriteLine("=== BST INSERT & SEARCH DEMO ===");
        root = insertRecursion(root, 5);
        root = insertRecursion(root, 6);
        root = insertRecursion(root, 4);
        root = insertRecursion(root, 3);
        
        printInOrder(root);
    }
}
